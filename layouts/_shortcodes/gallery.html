"Name" }} {{- if eq .ResourceType "image" }} {{- $img := . | images.Filter images.AutoOrient -}} {{- $small := $img.Fill (printf "%s webp q78 Center" $thumb) -}} {{- $large := $img.Fit (printf "%s webp q82" $full) -}} {{- $caption := "" -}} {{- if ne .Title .Name }}{{ $caption = .Title }}{{ end -}} {{- end }} {{- end }}
